T = 10
for test_case in range(1, T + 1):
    N = int(input())
    board = [input() for _ in range(8)]
    answer = 0
    for i in range(8):
        row = board[i]
        col = ''.join([board[j][i] for j in range(8)])
        for k in range(8 - N + 1):
            row_str = row[k : k + N]
            col_str = col[k : k + N]
            if row_str == row_str[::-1]:
                answer += 1
            if col_str == col_str[::-1]:
                answer += 1
                
    print(f"#{test_case} {answer}")
