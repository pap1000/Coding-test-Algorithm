T = 10

for test_case in range(1, T + 1):
    N = int(input())
    table = [list(map(int, input().split())) for _ in range(N)]
    
    answer = 0
    
    for i in range(N):
        check_N = False
        
        for j in range(N):
            if table[j][i] == 1:
                check_N = True
                
            elif table[j][i] == 2 and check_N:
                answer += 1
                check_N = False
                
    print(f"#{test_case} {answer}")
